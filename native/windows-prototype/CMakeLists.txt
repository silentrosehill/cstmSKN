cmake_minimum_required(VERSION 3.24)
project(cstmskn_injector_prototype LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The prototype requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
add_compile_options(/W4 /WX /permissive- /utf-8 /guard:cf)
add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
add_library(cstm-overlay-prototype SHARED overlay.cpp)
target_link_libraries(cstm-overlay-prototype PRIVATE user32 psapi)
add_executable(cstm-injector-fixture WIN32 fixture.cpp)
target_link_libraries(cstm-injector-fixture PRIVATE user32)
add_executable(cstm-injector-prototype host.cpp)
target_link_libraries(cstm-injector-prototype PRIVATE user32)
enable_testing()
add_test(NAME native-injector-lifecycle COMMAND cstm-injector-prototype --self-test)
set_tests_properties(native-injector-lifecycle PROPERTIES TIMEOUT 90)
find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
add_test(NAME native-injector-load-failure COMMAND "${POWERSHELL_EXECUTABLE}"
  -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/test-load-failure.ps1"
  -BinaryDirectory "$<TARGET_FILE_DIR:cstm-injector-prototype>")
set_tests_properties(native-injector-load-failure PROPERTIES TIMEOUT 40)
